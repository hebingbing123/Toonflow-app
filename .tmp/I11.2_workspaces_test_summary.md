## Testing Pattern

All tests follow the established pattern from `backend/src/settings/admin_console/mod.rs`:

### English Tests
```rust
#[tokio::test]
async fn error_name_response_en() {
    let err = helper_i18n("English message", "中文消息");
    let resp = err.into_response();
    
    let bytes = to_bytes(resp.into_body(), 16 * 1024).await.expect("body bytes");
    let json: serde_json::Value = serde_json::from_slice(&bytes).expect("json body");
    
    assert_eq!(json.get("status").and_then(|v| v.as_u64()), Some(expected_status));
    assert_eq!(json.get("code").and_then(|v| v.as_str()), Some("expected_code"));
    assert_eq!(json.get("message").and_then(|v| v.as_str()), Some("English message"));
}
```

### Chinese Tests
```rust
#[tokio::test]
async fn error_name_response_zh() {
    let resp = REQUEST_LOCALE
        .scope(ApiLocale::Zh, async {
            let err = helper_i18n("English message", "中文消息");
            err.into_response()
        })
        .await;
    
    let bytes = to_bytes(resp.into_body(), 16 * 1024).await.expect("body bytes");
    let json: serde_json::Value = serde_json::from_slice(&bytes).expect("json body");
    
    assert_eq!(json.get("status").and_then(|v| v.as_u64()), Some(expected_status));
    assert_eq!(json.get("code").and_then(|v| v.as_str()), Some("expected_code"));
    assert_eq!(json.get("message").and_then(|v| v.as_str()), Some("中文消息"));
}
```
